# The development-only dependencies of the Python side.
#
# The package itself is **standard library only at runtime** (`pyproject.toml`
# says so too): everything listed here is a tool the *gate* and the *test
# suite* use, never something the runner needs at runtime.
#
#   pip install -r requirements-dev.txt
#
# Pinned: a gate that changes its mind between runs is not a gate.

# The build system. Pinned like the rest: a build that changes its mind
# between runs is not a build either. These are the versions this repository
# is verified with (the CI installs them from this file).
meson==1.11.2
ninja==1.13.0

# The type checker. `scripts/typecheck.py` runs it over `scripts/scenario` and
# exits non-zero on an error — the configuration lives in `pyproject.toml`
# (`[tool.mypy]`).
mypy==1.14.1

# The style gate's clang-tidy (`scripts/clang-tidy-fix.py --check`). Pinned to
# the LLVM release the gated groups were baselined with: a gate that changes
# its mind between runs is not a gate. The package ships the LLVM binary; the
# CI puts it first on PATH so the gate cannot silently run the runner's own
# (unpinned) LLVM.
clang-tidy==22.1.8

# The MCP SDK. Only the *reference client* test needs it (it drives the MCP
# server with the real client); that test is skipped when the package is
# absent, because the server itself is standard library only.
mcp>=2.0,<3
